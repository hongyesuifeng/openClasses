```csharp
public class CombatStateSynchronizer
{
    // 同步阶段
    public enum SyncPhase
    {
        None,
        WaitingForPlayerData,
        WaitingForRngSync,
        Completed
    }

        // 广播玩家数据请求

            _netService.Broadcast(message);
        }
    }
```
```csharp
}

    public void Deserialize(BitReader reader)
    {
        PlayerIndex = reader.ReadInt();
        PlayerData = new SerializablePlayer();
        PlayerData.Deserialize(reader);
    }
}

private void OnSyncPlayerData(SyncPlayerDataMessage message)
```

### 2.4 行动队列同步

```csharp
public class ActionQueueSynchronizer
{
    private readonly Queue<GameAction> _pendingActions = new();

    // 请求执行行动
    public async Task RequestAction(GameAction action)
    {
        if (_netService.IsHost)
        {
            // 主机直接执行
            await ExecuteAction(action);
        }
        else
        {
            // 客户端发送请求
            var message = new RequestEnqueueActionMessage(action);
            _netService.SendMessage(message);

            // 等待主机确认
            await WaitForActionConfirmation(action.Id);
        }
    }

    // 主机处理行动请求
    private void OnRequestEnqueueAction(RequestEnqueueActionMessage message)
    {
        if (_netService.IsHost)
        {
            var action = message.Action;

            // 验证行动有效性
            if (ValidateAction(action))
            {
                // 加入队列
                EnqueueAction(action);

                // 广播确认
                var confirm = new ActionEnqueuedMessage(action);
                _netService.Broadcast(confirm);
            }
        }
    }
}
```

### 3.2 消息总线

```csharp
public class NetMessageBus
{
    private readonly Dictionary<Type, List<Action<INetMessage>>> _handlers = new();
    private readonly NetTypeCache _typeCache;

    // 注册消息处理器
    public void RegisterHandler<T>(Action<T> handler) where T : INetMessage
    {
        var type = typeof(T);
        if (!_handlers.ContainsKey(type))
        {
            _handlers[type] = new List<Action<INetMessage>>();
        }
        _handlers[type].Add(msg => handler((T)msg));
    }

    // 注销消息处理器
    public void UnregisterHandler<T>(Action<T> handler) where T : INetMessage
    {
        var type = typeof(T);
        if (_handlers.TryGetValue(type, out var handlers))
        {
            handlers.RemoveAll(h => h.Equals(handler));
        }
    }

    // 处理接收到的消息
    public void ProcessMessage(byte[] data)
    {
        using var reader = new BitReader(data);

        // 读取消息类型 ID
        var typeId = reader.ReadInt();
        var messageType = _typeCache.GetType(typeId);

        // 反序列化消息
        var message = (INetMessage)Activator.CreateInstance(messageType);
        message.Deserialize(reader);

        // 分发到处理器
        if (_handlers.TryGetValue(messageType, out var handlers))
        {
            foreach (var handler in handlers)
            {
                handler(message);
            }
        }
    }
}
```

### 3.3 序列化优化

```csharp
// 位级序列化（节省带宽）
public class BitWriter
{
    private readonly List<byte> _data = new();
    private int _currentBit;

    public void WriteInt(int value, int bits = 32)
    {
        for (int i = 0; i < bits; i++)
        {
            WriteBit((value & (1 << i)) != 0);
        }
    }

    public void WriteBool(bool value) => WriteBit(value);

    public byte[] ToArray() => _data.ToArray();
}

public class BitReader
{
    private readonly byte[] _data;
    private int _currentBit;

    public int ReadInt(int bits = 32)
    {
        int value = 0;
        for (int i = 0; i < bits; i++)
        {
            if (ReadBit())
                value |= (1 << i);
        }
        return value;
    }

    public bool ReadBool() => ReadBit();
}
```

## 4. Steam 集成

### 4.1 Steam 网络层

```csharp
public class SteamHost : INetTransport
{
    private HSteamNetConnection _listenSocket;
    private readonly List<HSteamNetConnection> _connections = new();

    public async Task<bool> Start(ushort port)
    {
        // 创建监听 socket
        var options = new SteamNetworkingConfigValue_t[]
        {
            new() { m_eValue = ESteamNetworkingConfigValue.k_ESteamNetworkingConfig_TimeoutInitial, m_val = { m_int32 = 20000 } }
        };

        _listenSocket = SteamNetworkingSockets.CreateListenSocketP2P(0, options.Length, options);

        return _listenSocket != HSteamNetConnection.Invalid;
    }

    public void Send(HSteamNetConnection connection, byte[] data)
    {
        SteamNetworkingSockets.SendMessageToConnection(
            connection,
            data,
            data.Length,
            Constants.k_nSteamNetworkingSend_Reliable
        );
    }

    public void PollMessages()
    {
        var messages = new SteamNetworkingMessage_t[64];
        int count = SteamNetworkingSockets.ReceiveMessagesOnConnection(_listenSocket, messages, 64);

        for (int i = 0; i < count; i++)
        {
            var message = messages[i];
            OnMessageReceived?.Invoke(message.m_conn, message.m_pData, message.m_cbSize);
            message.Release();
        }
    }
}
```

### 4.2 Steam 大厅系统

```csharp
public class SteamLobbyManager
{
    private CSteamID _currentLobby;

    // 创建大厅
    public async Task<CSteamID> CreateLobby(int maxPlayers)
    {
        var result = await SteamMatchmaking.CreateLobbyAsync(ELobbyType.k_ELobbyTypeFriendsOnly, maxPlayers);
        if (result.Success)
        {
            _currentLobby = result.LobbyID;

            // 设置大厅数据
            SteamMatchmaking.SetLobbyData(_currentLobby, "name", "Slay the Spire 2 Game");
            SteamMatchmaking.SetLobbyData(_currentLobby, "version", GameVersion.ToString());
        }
        return _currentLobby;
    }

    // 加入大厅
    public async Task<bool> JoinLobby(CSteamID lobbyId)
    {
        var result = await SteamMatchmaking.JoinLobbyAsync(lobbyId);
        if (result.Success)
        {
            _currentLobby = lobbyId;
            OnLobbyJoined?.Invoke(lobbyId);
            return true;
        }
        return false;
    }

    // 获取大厅成员
    public List<CSteamID> GetLobbyMembers()
    {
        var members = new List<CSteamID>();
        int count = SteamMatchmaking.GetNumLobbyMembers(_currentLobby);

        for (int i = 0; i < count; i++)
        {
            members.Add(SteamMatchmaking.GetLobbyMemberByIndex(_currentLobby, i));
        }

        return members;
    }
}
```

### 5.1 质量追踪器

```csharp
public class NetQualityTracker
{
    private readonly List<long> _pingHistory = new();
    private DateTime _lastHeartbeat;

    public void SendHeartbeat()
    {
        var message = new HeartbeatRequestMessage
        {
            Timestamp = DateTime.UtcNow.Ticks
        };
        _netService.Broadcast(message);
    }

    public void OnHeartbeatResponse(HeartbeatResponseMessage message)
    {
        var latency = DateTime.UtcNow.Ticks - message.OriginalTimestamp;
        _pingHistory.Add(latency / TimeSpan.TicksPerMillisecond);

        // 保持历史记录限制
        while (_pingHistory.Count > 100)
        {
            _pingHistory.RemoveAt(0);
        }

        // 记录统计
        LogStatistics();
    }

    public int AveragePing => (int)_pingHistory.Average();
    public int MinPing => _pingHistory.Min();
    public int MaxPing => _pingHistory.Max();
}
```

### 5.2 连接状态管理

```csharp
public enum ConnectionState
{
    Disconnected,
    Connecting,
    Connected,
    Reconnecting,
    Timeout
}

public class ConnectionManager
{
    private ConnectionState _state;
    private DateTime _lastActivity;
    private readonly TimeSpan _timeoutDuration = TimeSpan.FromSeconds(20);

    public void Update()
    {
        if (_state == ConnectionState.Connected)
        {
            var sinceLastActivity = DateTime.UtcNow - _lastActivity;
            if (sinceLastActivity > _timeoutDuration)
            {
                OnTimeout();
            }
        }
    }

    private void OnTimeout()
    {
        _state = ConnectionState.Timeout;
        OnConnectionLost?.Invoke();
    }

    public void ResetTimeout()
    {
        _lastActivity = DateTime.UtcNow;
    }
}
```

## 6. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 多人游戏系统的设计亮点：**

1. **分层架构**：网络层、同步层、游戏层清晰分离，便于维护
2. **状态同步**：RNG 同步确保所有玩家看到相同的结果
3. **行动队列同步**：确保行动按正确顺序执行
4. **断线重连**：支持玩家重连后恢复游戏状态
5. **Steam 集成**：利用 Steam P2P 和大厅系统简化联机
6. **位级序列化**：优化带宽使用
`─────────────────────────────────────────────────`
