### 1.3 路径生成算法

```csharp
private void PathGenerate(MapPoint current)
{
    while (current.coord.row < GetRowCount() - 1)
    {
        MapCoord nextCoord = GenerateNextCoord(current);

        MapPoint? nextPoint = GetPoint(nextCoord);
        if (nextPoint == null)
        {
            nextPoint = CreatePoint(nextCoord);
        }

        current.AddChildPoint(nextPoint);
        current = nextPoint;
    }
}

private MapCoord GenerateNextCoord(MapPoint current)
{
    // 随机选择左/中/右三个方向
    int[] directions = { -1, 0, 1 };
    rng.Shuffle(directions);

    foreach (int dir in directions)
    {
        int targetX = current.coord.col + dir;

        // 检查边界
        if (targetX < 0 || targetX >= 7) continue;

        // 检查是否有无效交叉
        if (!HasInvalidCrossover(current, targetX))
        {
            return new MapCoord { col = targetX, row = current.coord.row + 1 };
        }
    }

    throw new InvalidOperationException("Cannot find next node");
}
```

### 1.4 无效交叉检测

```csharp
private bool HasInvalidCrossover(MapPoint current, int targetX)
{
    // 检测路径交叉情况
    // 例如：从(2,1)到(3,2) 不应该与 从(3,1)到(2,2) 交叉

    int currentX = current.coord.col;
    int nextRow = current.coord.row + 1;

    // 检查相邻列的节点是否有反向连接
    if (targetX > currentX)  // 向右移动
    {
        var rightPoint = GetPoint(new MapCoord(currentX + 1, current.coord.row));
        if (rightPoint?.HasChildAtCol(currentX) == true)
            return true;  // 存在交叉
    }
    else if (targetX < currentX)  // 向左移动
    {
        var leftPoint = GetPoint(new MapCoord(currentX - 1, current.coord.row));
        if (leftPoint?.HasChildAtCol(currentX) == true)
            return true;  // 存在交叉
    }

    return false;
}
```

### 2.4 节点连接结构

```csharp
public class MapPoint : IComparable<MapPoint>
{
    public readonly MapCoord coord;           // 坐标
    public readonly HashSet<MapPoint> parents;   // 父节点集合
    public HashSet<MapPoint> Children { get; }   // 子节点集合
    public MapPointType PointType { get; set; }  // 节点类型
    public bool IsVisited { get; set; }          // 是否已访问

    // 添加子节点
    public void AddChildPoint(MapPoint child)
    {
        Children.Add(child);
        child.parents.Add(this);
    }
}

public struct MapCoord
{
    public int col;  // 列（0-6）
    public int row;  // 行
}
```

### 3.2 修复节点类型

```csharp
public static bool RepairPrunedPointTypes(
    ActMap map,
    MapPointTypeCounts targetCounts,
    Rng rng,
    Func<MapPointType, MapPoint, bool> isValidType)
{
    // 确保各类型节点数量正确
    var currentCounts = CountPointTypes(map);
    var differences = CalculateDifferences(currentCounts, targetCounts);

    foreach (var diff in differences)
    {
        if (diff.count > 0)  // 需要添加
        {
            AddPointType(map, diff.type, diff.count, rng, isValidType);
        }
        else if (diff.count < 0)  // 需要移除
        {
            RemovePointType(map, diff.type, -diff.count, rng);
        }
    }

    return differences.Any();
}
```

### 4.3 事件流程示例

```csharp
// 示例：深渊浴场事件
public sealed class AbyssalBaths : EventModel
{
    protected override IEnumerable<DynamicVar> CanonicalVars => new DynamicVar[]
    {
        new MaxHpVar(2m),           // 最大生命值变化
        new DamageVar(3m, ValueProp.Unblockable | ValueProp.Unpowered)  // 不可格挡伤害
    };

    protected override IReadOnlyList<EventOption> GenerateInitialOptions()
    {
        return new[]
        {
            // 选项1：浸泡（受伤但获得最大生命值）
            new EventOption(this, OnImmerse, "ABYSSAL_BATHS.pages.INITIAL.options.IMMERSE")
                .ThatDoesDamage(DynamicVars.Damage.IntValue - DynamicVars.MaxHp.BaseValue),

            // 选项2：放弃
            new EventOption(this, OnAbstain, "ABYSSAL_BATHS.pages.INITIAL.options.ABSTAIN")
        };
    }

    private async Task OnImmerse()
    {
        await PlayerCmd.LoseHp(Owner, DynamicVars.Damage.IntValue);
        await PlayerCmd.GainMaxHp(Owner, DynamicVars.MaxHp.IntValue);
        SetEventFinished();
    }

    private async Task OnAbstain()
    {
        SetEventFinished();
    }
}
```

## 7. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 地图与事件系统的设计亮点：**

1. **随机且有约束的地图生成**：随机路径 + 严格限制规则，确保每局游戏既有变化又保持平衡
2. **节点类型限制**：避免同类型节点过于集中，保证游戏节奏
3. **事件选项系统**：通过链式配置创建复杂的事件选项，代码可读性高
4. **动态变量复用**：事件与卡牌共用 DynamicVar 系统
`─────────────────────────────────────────────────`
