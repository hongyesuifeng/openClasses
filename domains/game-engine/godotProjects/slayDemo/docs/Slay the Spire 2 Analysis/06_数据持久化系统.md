### 2.4 运行时保存流程

```csharp
public async Task SaveRun(AbstractRoom? preFinishedRoom, bool saveProgress = true)
{
    // 等待当前保存完成
    if (CurrentRunSaveTask != null)
    {
        await CurrentRunSaveTask;
    }

    // 开始批量保存
    using (BeginSaveBatch())
    {
        // 保存进度文件
        if (saveProgress)
        {
            SaveProgressFile();
        }

        // 保存运行数据
        try
        {
            CurrentRunSaveTask = _runSaveManager.SaveRun(preFinishedRoom);
            await CurrentRunSaveTask;
        }
        finally
        {
            CurrentRunSaveTask = null;
        }
    }
}
```

public class UnlockState

    public IEnumerable<CardModel> Cards { get; }

    public IEnumerable<PotionModel> Potions { get; }

        return _unlockedEpochIds.Contains(EpochModel.GetId<T>());
    }

    {
        _unlockedEpochIds.Add(epochId);
        SaveProgress();
    }
}

## 4. 数据序列化

### 4.1 JSON 序列化系统

```csharp
public static class JsonSerializationUtility
{
    // 序列化选项
    public static JsonSerializerOptions Options { get; } =
        new JsonSerializerOptions(MegaCritSerializerContext.DefaultGeneratedSerializerOptions)
        {
            Encoder = JavaScriptEncoder.UnsafeRelaxedJsonEscaping,
            TypeInfoResolver = MegaCritSerializerContext.Default.WithAddedModifier(AlphabetizeProperties)
        };

    // 序列化为 JSON
    public static string ToJson<T>(T obj) where T : ISaveSchema
    {
        return JsonSerializer.Serialize(obj, GetTypeInfo<T>());
    }

    // 从 JSON 反序列化
    public static ReadSaveResult<T> FromJson<T>(string json)
        where T : ISaveSchema, new()
    {
        try
        {
            var data = JsonSerializer.Deserialize<T>(json, GetTypeInfo<T>());
            return ReadSaveResult<T>.Success(data!);
        }
        catch (Exception ex)
        {
            return ReadSaveResult<T>.Failure(ex.Message);
        }
    }
}
```

### 4.2 Schema 版本控制

```csharp
public interface ISaveSchema
{
    int SchemaVersion { get; set; }

// 版本迁移示例
public static class SaveMigrations
{
    public static SerializableRun Migrate(SerializableRun data)
    {
        // 从旧版本迁移到新版本
        while (data.SchemaVersion < CurrentSchemaVersion)
        {
            data = MigrateToNextVersion(data);
            data.SchemaVersion++;
        }
        return data;
    }

    private static SerializableRun MigrateToNextVersion(SerializableRun data)
    {
        // 根据版本号执行具体迁移逻辑
        return data.SchemaVersion switch
        {
            1 => MigrateFromV1ToV2(data),
            2 => MigrateFromV2ToV3(data),
            // ...
            _ => data
        };
    }
}
```

public static ISaveStore CreateSaveStore()
{
    var localStore = new LocalSaveStore();

    if (SteamInitializer.Initialized)
    {
        var cloudStore = new SteamRemoteSaveStore();
        var cloudSaveStore = new CloudSaveStore(localStore, cloudStore);

### 5.3 多设备同步

```csharp
public class CloudSaveStore : ISaveStore
{
    private readonly ISaveStore _localStore;
    private readonly IRemoteSaveStore _cloudStore;

    public async Task Save<T>(string key, T data) where T : ISaveSchema
    {
        // 保存到本地
        await _localStore.Save(key, data);

        // 上传到云端
        await _cloudStore.Upload(key, data);
    }

    public async Task<ReadSaveResult<T>> Load<T>(string key)
        where T : ISaveSchema, new()
    {
        // 检查云端是否有更新
        var cloudTimestamp = await _cloudStore.GetLastModified(key);
        var localTimestamp = await _localStore.GetLastModified(key);

        if (cloudTimestamp > localTimestamp)
        {
            // 下载云存档
            var cloudData = await _cloudStore.Download<T>(key);
            await _localStore.Save(key, cloudData);
            return ReadSaveResult<T>.Success(cloudData);
        }

        // 使用本地存档
        return await _localStore.Load<T>(key);
    }
}
```

## 6. 错误处理与恢复

### 6.1 错误类型

```csharp
public enum SaveErrorType
{
    None,
    FileNotFound,
    InvalidFormat,
    SchemaMismatch,
    MigrationFailed,
    CloudSyncFailed,
    WriteFailed,
    ReadFailed
}

public class ReadSaveResult<T>
{
    public bool IsSuccess { get; }
    public T? Data { get; }
    public SaveErrorType ErrorType { get; }
    public string? ErrorMessage { get; }

    public static ReadSaveResult<T> Success(T data) => new(true, data);
    public static ReadSaveResult<T> Failure(string message, SaveErrorType type = SaveErrorType.ReadFailed)
        => new(false, default, type, message);
}
```

### 6.2 恢复机制

```csharp
public async Task LoadRun()
{
    var result = await _runSaveManager.LoadRun();

    if (!result.IsSuccess)
    {
        switch (result.ErrorType)
        {
            case SaveErrorType.SchemaMismatch:
            case SaveErrorType.MigrationFailed:
                // 尝试修复存档
                var repaired = await AttemptRepair(result);
                if (repaired.IsSuccess)
                {
                    LoadRunState(repaired.Data);
                    return;
                }
                break;

            case SaveErrorType.FileNotFound:
                // 没有存档，创建新运行
                CreateNewRun();
                return;

            default:
                // 显示错误并重置
                ShowSaveError(result.ErrorMessage);
                CreateNewRun();
                return;
        }
    }

    LoadRunState(result.Data);
}
```

## 7. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 数据持久化系统的设计亮点：**

1. **分层架构**：不同类型数据分离管理，职责清晰
2. **Schema 版本控制**：支持存档迁移，保证向后兼容
3. **异步保存**：非阻塞保存，不影响游戏流畅度
4. **云同步**：Steam 云集成，多设备无缝同步
5. **完善错误处理**：各种异常情况都有对应的恢复策略
`─────────────────────────────────────────────────`
