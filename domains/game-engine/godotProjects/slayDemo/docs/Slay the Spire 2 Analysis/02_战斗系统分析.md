### 1.3 双阶段结束机制

```csharp
// 第一阶段：准备结束
public async Task EndPlayerTurnPhaseOneInternal()
{
    // 1. 处理手牌效果（如虚无牌消耗）
    await ProcessEtherealCards();

    // 2. 触发回合结束钩子
    await Hook.AfterTurnEnd(choiceContext, CombatSide.Player);

    // 3. 检查胜利条件
    if (CheckVictoryCondition())
    {
        await EndCombatVictory();
        return;
    }
}

// 第二阶段：切换到敌人回合
public async Task AfterAllPlayersReadyToBeginEnemyTurn()
{
    // 1. 丢弃手牌
    await DiscardAllCards();

    // 2. 设置敌人回合标志
    IsEnemyTurnStarted = true;

    // 3. 执行敌人行动
    await ExecuteEnemyTurn();

    // 4. 开始下一回合
    RoundNumber++;
    await StartTurn();
}
```

public class PlayerCombatState

## 3. 伤害计算系统

### 3.1 伤害计算流程

```csharp
public async Task<DamageResult> DealDamage(
    Creature target, Creature dealer, decimal baseAmount,
    ValueProp props, CardModel? cardSource)
{
    // 1. 基础伤害修改（加法）
    decimal amount = baseAmount;
    amount += Hook.ModifyDamageAdditive(target, dealer, amount, props);

    // 2. 伤害修改（乘法）
    amount *= Hook.ModifyDamageMultiplicative(target, dealer, amount, props);

    // 3. 最终修改
    amount = Hook.ModifyDamage(runState, combatState, target, dealer,
        amount, props, cardSource, ModifyDamageHookType.All);

    // 4. 计算格挡
    decimal blocked = target.DamageBlockInternal(amount, props);
    decimal unblocked = Math.Max(amount - blocked, 0);

    // 5. 应用伤害
    DamageResult result = await target.LoseHpInternal(unblocked, props, dealer);

    return result;
}
```

### 3.2 伤害计算公式

```
最终伤害 = ((基础伤害 + 力量) × 伤害倍率) - 格挡

其中：
- 力量：StrengthPower 的加法修正
- 伤害倍率：
  - 虚弱(Weak)：0.75（自己攻击时）
  - 易伤(Vulnerable)：1.5（被攻击时）
- 格挡：目标的当前格挡值
```

### 3.3 值属性（ValueProp）

```csharp
[Flags]
public enum ValueProp
{
    None          = 0,
    Move          = 1,    // 普通攻击，受力量/虚弱影响
    Unpowered     = 2,    // 不受力量影响
    Unblockable   = 4,    // 不可格挡
    SkipHurtAnim  = 8,    // 跳过受伤动画
    Thorn         = 16,   // 荆棘伤害
}
```

## 4. 格挡系统

### 4.1 格挡机制

```csharp
public class Creature
{
    private int _block;

    public int Block { get; private set; }

    // 受到伤害时计算格挡
    public decimal DamageBlockInternal(decimal amount, ValueProp props)
    {
        // 不可格挡伤害直接返回
        if (props.HasFlag(ValueProp.Unblockable))
            return 0;

        // 计算格挡量
        decimal blocked = Math.Min(Block, amount);
        Block -= (int)blocked;

        return blocked;
    }

    // 获得格挡
    public void GainBlockInternal(decimal amount)
    {
        Block = (int)Math.Min(Block + amount, 999999999);
    }
}
```

### 4.2 格挡清除

// 回合结束时清除格挡

### 5.3 核心状态效果

#### Strength（力量）

```csharp
public sealed class StrengthPower : PowerModel
{
    public override PowerType Type => PowerType.Buff;
    public override PowerStackType StackType => PowerStackType.Counter;

    // 增加攻击伤害
    public override decimal ModifyDamageAdditive(
        Creature? target, decimal amount, ValueProp props,
        Creature? dealer, CardModel? cardSource)
    {
        // 只有自己攻击时生效
        if (dealer == Owner && props.IsPoweredAttack())
            return Amount;  // 每层力量增加等量伤害
        return 0m;
    }
}
```

#### Weak（虚弱）

```csharp
public sealed class WeakPower : PowerModel
{
    public override PowerType Type => PowerType.Debuff;
    public override PowerStackType StackType => PowerStackType.Counter;

    // 减少攻击伤害
    public override decimal ModifyDamageMultiplicative(
        Creature? target, decimal amount, ValueProp props,
        Creature? dealer, CardModel? cardSource)
    {
        // 自己攻击时伤害降低25%
        if (dealer == Owner && props.IsPoweredAttack())
            return 0.75m;
        return 1m;
    }

    // 回合结束时减少层数
    public override async Task AfterTurnEnd(...)
    {
        if (side == CombatSide.Player)
        {
            await PowerCmd.TickDownDuration(this);
        }
    }
}
```

#### Vulnerable（易伤）

```csharp
public sealed class VulnerablePower : PowerModel
{
    public override PowerType Type => PowerType.Debuff;
    public override PowerStackType StackType => PowerStackType.Counter;

    // 增加受到的伤害
    public override decimal ModifyDamageMultiplicative(
        Creature? target, decimal amount, ValueProp props,
        Creature? dealer, CardModel? cardSource)
    {
        // 被攻击时伤害增加50%
        if (target == Owner && props.IsPoweredAttack())
            return 1.5m;
        return 1m;
    }
}
```

### 5.4 状态效果应用

```csharp
// 应用能力
public static async Task ApplyPower<T>(
    Creature target, int amount, Creature? source = null)
    where T : PowerModel
{
    // 检查是否已存在
    var existing = target.GetPower<T>();
    if (existing != null)
    {
        if (existing.StackType == PowerStackType.Counter)
            existing.Amount += amount;
    }
    else
    {
        var power = CreatePower<T>(target, amount, source);
        target.AddPower(power);
        await Hook.AfterPowerApplied(combatState, power);
    }
}
```

## 8. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 战斗系统的设计亮点：**

1. **清晰的伤害计算链**：加法→乘法→格挡→最终伤害，每步都可被Hook修改
2. **状态效果系统**：通过PowerModel的虚方法实现高度可扩展的效果系统
3. **意图系统**：玩家可以预判敌人行动，增加策略深度
4. **异步战斗**：完整支持多人游戏同步的异步架构
`─────────────────────────────────────────────────`
