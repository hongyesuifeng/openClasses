## 2. Hook 系统

### 2.1 Hook 类型设计

```csharp
// 运行状态 Hook
public delegate IEnumerable<AbstractModel> RunHookSubscriptionDelegate(RunState runState);

// 战斗状态 Hook
public delegate IEnumerable<AbstractModel> CombatHookSubscriptionDelegate(CombatState combatState);
```

### 2.2 Hook 订阅机制

```csharp
public static class ModHelper
{
    // Hook 订阅存储
    private static readonly Dictionary<string, RunHookSubscriptionDelegate> _runStateHooks = new();
    private static readonly Dictionary<string, CombatHookSubscriptionDelegate> _combatStateHooks = new();

    // 订阅运行状态 Hook
    public static void SubscribeForRunStateHooks(string id, RunHookSubscriptionDelegate del)
    {
        if (!_runStateHooks.ContainsKey(id))
        {
            _runStateHooks[id] = del;
        }
    }

    // 订阅战斗状态 Hook
    public static void SubscribeForCombatStateHooks(string id, CombatHookSubscriptionDelegate del)
    {
        if (!_combatStateHooks.ContainsKey(id))
        {
            _combatStateHooks[id] = del;
        }
    }
}
```

### 2.3 Hook 执行流程

```csharp
// 在 RunState 中的迭代执行
public class RunState
{
    public IEnumerable<AbstractModel> GetHookResults()
    {
        // 按字母顺序执行所有订阅的 Hook
        foreach (var kvp in ModHelper._runStateHooks.OrderBy(x => x.Key))
        {
            foreach (var result in kvp.Value(this))
            {
                yield return result;
            }
        }
    }
}

// 在 CombatState 中的迭代执行
public class CombatState
{
    public IEnumerable<AbstractModel> GetHookResults()
    {
        foreach (var kvp in ModHelper._combatStateHooks.OrderBy(x => x.Key))
        {
            foreach (var result in kvp.Value(this))
            {
                yield return result;
            }
        }
    }
}
```

### 2.4 Hook 使用示例

```csharp
// 模组初始化时订阅 Hook
public class MyModInitializer
```

## 3. 内容注入机制

### 3.1 模型池扩展 API

```csharp
public static class ModHelper
{
    // 泛型 API
    public static void AddModelToPool<TPoolType, TModelType>()
        where TPoolType : AbstractModel, IPoolModel
        where TModelType : AbstractModel
    {
        var pool = ModelDb.GetPool<TPoolType>();
        pool.AddModel(CreateModel<TModelType>());
    }

    // 非泛型 API
    public static void AddModelToPool(Type poolType, Type modelType)
    {
        var pool = ModelDb.GetPool(poolType);
        var model = CreateModel(modelType);
        pool.AddModel(model);
    }
}
```

### 3.2 内容注入流程

```
┌─────────────────────────────────────────────────────────────┐
│                  1. 模组初始化阶段                           │
│  - 调用 ModInitializer 标记的方法                            │
│  - 调用 AddModelToPool 注册新内容                            │
│  - 订阅 Hook                                                │
└────────────────────────┬────────────────────────────────────┘
                         ↓
┌─────────────────────────────────────────────────────────────┐
│                  2. 模组池冻结                               │
│  - 游戏初始化完成后冻结所有池                                │
│  - 禁止添加新内容                                            │
└────────────────────────┬────────────────────────────────────┘
                         ↓
┌─────────────────────────────────────────────────────────────┐
│                  3. 内容合并                                 │
│  - ConcatModelsFromMods 将模组内容注入游戏                   │
│  - 在卡牌池、遗物池、药水池中使用                             │
└─────────────────────────────────────────────────────────────┘
```

### 3.3 池合并实现

```csharp
public static IEnumerable<TModelType> ConcatModelsFromMods<TModelType>(
    IPoolModel pool,
    IEnumerable<TModelType> baseModels)
{
    // 返回基础内容 + 模组内容
    foreach (var model in baseModels)
    {
        yield return model;
    }

    // 添加模组注入的内容
    foreach (var hookResult in GetHookResults())
    {
        if (hookResult is TModelType typedResult)
        {
            yield return typedResult;
        }
    }
}
```

## 4. 模组加载流程

### 4.1 加载入口

```csharp
public class ModManager
{
    public async Task LoadMods()
    {
        // 1. 发现所有模组
        var mods = await DiscoverMods();

        // 2. 解析依赖关系
        var sortedMods = SortByDependencies(mods);

        // 3. 按顺序加载
        foreach (var mod in sortedMods)
        {
            await LoadMod(mod);
        }
    }
}
```

### 4.2 模组发现

```csharp
private async Task<List<Mod>> DiscoverMods()
{
    var mods = new List<Mod>();

    // 1. 扫描本地模组目录
    var localMods = await ScanDirectory("mods/");
    mods.AddRange(localMods);

    // 2. 扫描 Steam 创意工坊
    if (SteamInitializer.Initialized)
    {
        var workshopMods = await ScanSteamWorkshop();
        mods.AddRange(workshopMods);
    }

    return mods;
}
```

### 4.3 依赖排序

```csharp
private List<Mod> SortByDependencies(List<Mod> mods)
{
    // 使用拓扑排序处理依赖
    var dependencyCount = new Dictionary<Mod, int>();
    var dependents = new Dictionary<Mod, List<Mod>>();

    // 构建依赖图
    foreach (var mod in mods)
    {
        dependencyCount[mod] = mod.Dependencies.Count;
        foreach (var depId in mod.Dependencies)
        {
            var dep = mods.FirstOrDefault(m => m.Id == depId);
            if (dep != null)
            {
                if (!dependents.ContainsKey(dep))
                    dependents[dep] = new List<Mod>();
                dependents[dep].Add(mod);
            }
        }
    }

    // Kahn 算法
    var queue = new Queue<Mod>(mods.Where(m => dependencyCount[m] == 0));
    var result = new List<Mod>();

    while (queue.Count > 0)
    {
        var mod = queue.Dequeue();
        result.Add(mod);

        if (dependents.TryGetValue(mod, out var dependentMods))
        {
            foreach (var dependent in dependentMods)
            {
                dependencyCount[dependent]--;
                if (dependencyCount[dependent] == 0)
                    queue.Enqueue(dependent);
            }
        }
    }

    // 检查循环依赖
    if (result.Count < mods.Count)
    {
        // 报告循环依赖错误
    }

    return result;
}
```

### 4.4 加载单个模组

```csharp
private async Task LoadMod(Mod mod)
{
    try
    {
        // 1. 加载 DLL
        if (!string.IsNullOrEmpty(mod.DllPath))
        {
            mod.LoadedAssembly = Assembly.LoadFrom(mod.DllPath);
        }

        // 2. 加载资源包
        if (!string.IsNullOrEmpty(mod.PckPath))
        {
            ProjectSettings.LoadResourcePack(mod.PckPath);
        }

        // 3. 执行初始化
        InitializeMod(mod);

        mod.LoadState = ModLoadState.Loaded;
    }
    catch (Exception ex)
    {
        mod.LoadState = ModLoadState.Failed;
        mod.LoadError = ex.Message;
    }
}
```

### 4.5 初始化方式

```csharp
private void InitializeMod(Mod mod)
{
    var assembly = mod.LoadedAssembly;
    if (assembly == null) return;

    // 方式一：查找 ModInitializer 属性
    foreach (var type in assembly.GetTypes())
    {
        var attr = type.GetCustomAttribute<ModInitializerAttribute>();
        if (attr != null)
        {
            var method = type.GetMethod(attr.MethodName);
            method?.Invoke(null, null);
            return;
        }
    }

    // 方式二：自动 Harmony Patch
    mod.HarmonyInstance = new Harmony($"{mod.Author}.{mod.Id}");
    mod.HarmonyInstance.PatchAll(assembly);
}
```

## 5. Steam 创意工坊集成

### 5.1 创意工坊发现

```csharp
private async Task<List<Mod>> ScanSteamWorkshop()
{
    var mods = new List<Mod>();

    // 获取已订阅项目
    uint numSubscribed = SteamUGC.GetNumSubscribedItems();
    var items = new PublishedFileId_t[numSubscribed];
    SteamUGC.GetSubscribedItems(items, numSubscribed);

    foreach (var item in items)
    {
        // 获取安装路径
        if (SteamUGC.GetItemInstallInfo(item, out var size, out var path, 256))
        {
            var mod = await LoadModFromDirectory(path, ModSource.SteamWorkshop);
            if (mod != null)
                mods.Add(mod);
        }
    }

    return mods;
}
```

## 7. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 模组系统的设计亮点：**

1. **Hook 钩子系统**：几乎任何游戏行为都可被模组修改，扩展性极强
2. **懒加载 Hook**：只在被迭代时执行，性能友好
3. **双重初始化**：支持 ModInitializer 和 Harmony Patch 两种方式
4. **依赖管理**：自动解析依赖关系，拓扑排序加载顺序
5. **Steam 集成**：完整的创意工坊支持，实时监控新安装模组
`─────────────────────────────────────────────────`
