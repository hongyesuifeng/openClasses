# Slay the Spire 2 UI与场景系统分析

> 本文档分析游戏的UI架构和场景组织，包括场景层次、UI组件设计和动画系统。

## 1. 场景架构

### 1.1 核心场景层次

```
game.tscn (根场景)
├── NSceneContainer          # 场景容器管理器
├── NGame                    # 游戏主控制器
├── AudioManager             # 音频管理器
├── ScreenShake              # 屏幕震动效果
└── FmodBankLoader           # 音效库加载器
    └── scene_container.tscn # 动态场景容器
        ├── main_menu.tscn   # 主菜单
        ├── combat_ui.tscn   # 战斗界面
        └── map_screen.tscn  # 地图界面
```

### 1.2 场景容器系统

```csharp
public partial class NSceneContainer : Control
{
    private Control? _currentScene;
    public Control? CurrentScene
    {
        get => _currentScene;
        private set
        {
            _currentScene = value;
            // 通知场景变化
            OnSceneChanged?.Invoke(value);
        }
    }

    // 设置当前场景
    public void SetCurrentScene(Control node)
    {
        // 清理旧场景
        foreach (Node child in GetChildren())
        {
            this.RemoveChildSafely(child);
            child.QueueFreeSafely();
        }

        // 加载新场景
        CurrentScene = node;
        if (node.GetParent() == null)
        {
            this.AddChildSafely(node);
        }
        else
        {
            node.Reparent(this);
        }
    }
}
```

public partial class NCard : Control
{
    // 卡牌数据
    public CardModel? CardModel { get; private set; }

    // 视觉组件
    private NCardFrame? _frame;
    private NCardRarityGlow? _rarityGlow;
    private Label? _energyLabel;
    private Label? _titleLabel;
    private Label? _descriptionLabel;

    // 初始化
    public void Initialize(CardModel card)
    {
        CardModel = card;

        // 设置颜色主题
        var theme = GetCardTheme(card.Type);
        _frame.SetTheme(theme);

        // 设置稀有度光效
        _rarityGlow.SetRarity(card.Rarity);

        // 设置文本
        _energyLabel.Text = card.EnergyCost.GetWithModifiers(CostModifiers.All).ToString();
        _titleLabel.Text = card.Title;
        _descriptionLabel.Text = card.Description;
    }

    // 获取卡牌主题
    private CardTheme GetCardTheme(CardType type)
    {
        return type switch
        {
            CardType.Attack => CardTheme.Attack,
            CardType.Skill => CardTheme.Skill,
            CardType.Power => CardTheme.Power,
            _ => CardTheme.Default
        };
    }
}

## 4. 地图UI系统

### 4.1 地图绘制系统

```csharp
// 使用 SubViewport 实现独立渲染
public class NMapDrawing : Control
{
    private SubViewport? _viewport;
    private CanvasItemMaterial? _blendMaterial;

    public void DrawMap(ActMap map)
    {
        // 清除旧绘制
        _viewport.RenderTargetClear();

        // 绘制路径连线
        foreach (var point in map.GetAllPoints())
        {
            DrawPointConnections(point);
        }

        // 绘制节点图标
        foreach (var point in map.GetAllPoints())
        {
            DrawMapPoint(point);
        }
    }

    private void DrawMapPoint(MapPoint point)
    {
        string iconName = GetIconName(point.PointType);
        var icon = ResourceLoader.Load<Texture2D>($"res://images/map/{iconName}.png");
        DrawTexture(icon, GetScreenPosition(point.coord));
    }

    private string GetIconName(MapPointType type)
    {
        return type switch
        {
            MapPointType.Monster => "map_monster",
            MapPointType.Elite => "map_elite",
            MapPointType.Treasure => "map_chest",
            MapPointType.Shop => "map_shop",
            MapPointType.RestSite => "map_rest",
            MapPointType.Boss => "map_boss",
            MapPointType.Ancient => "map_ancient",
            _ => "map_unknown"
        };
    }
}
```

public class NMapPoint : Control
{
    public MapPoint? MapPoint { get; private set; }

    public void Initialize(MapPoint point)
    {
        MapPoint = point;

        // 设置图标
        _icon.Texture = GetIconForType(point.PointType);

        // 设置可点击状态
        if (point.IsAccessible())
        {
            SetClickable();
        }
        else
        {
            SetDisabled();
        }

        // 已访问状态
        if (point.IsVisited)
        {
            SetVisited();
        }
    }

    public override void _GuiInput(InputEvent @event)
    {
        if (@event is InputEventMouseButton mouseButton)
        {
            if (mouseButton.Pressed && mouseButton.ButtonIndex == MouseButton.Left)
            {
                OnMapPointClicked?.Invoke(MapPoint);
            }
        }
    }
}

public class NTransition : Control
{
    public enum TransitionType
    {
        Fade,    // 渐变
        Fight,   // 战斗切换
        Instant  // 即时
    }

    public async Task Transition(TransitionType type, float duration)
    {
        switch (type)
        {
            case TransitionType.Fade:
                await FadeTransition(duration);
                break;
            case TransitionType.Fight:
                await FightTransition(duration);
                break;
            case TransitionType.Instant:
                InstantTransition();
                break;
        }
    }

    private async Task FadeTransition(float duration)
    {
        // 渐黑
        await TweenAlpha(0f, duration / 2);
        // 切换场景
        // ...
        // 渐显
        await TweenAlpha(1f, duration / 2);
    }
}

public class NCardFlyVfx : Node2D
{
    public static async Task FlyCard(CardModel card, Vector2 from, Vector2 to, float duration)
    {
        var vfx = CreateVfx<NCardFlyVfx>();
        vfx.Position = from;
        vfx.SetCard(card);

        // 创建飞行动画
        var tween = vfx.CreateTween();
        tween.TweenProperty(vfx, "position", to, duration)
             .SetEase(Tween.EaseType.Out)
             .SetTrans(Tween.TransitionType.Quad);

        await vfx.ToSignal(tween, "finished");
        vfx.QueueFree();
    }
}

## 7. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 UI系统的设计亮点：**

1. **模块化场景**：每个UI组件是独立场景，便于复用和维护
2. **SubViewport隔离**：地图绘制等复杂UI使用独立视口，避免渲染冲突
3. **主题系统**：统一的字体和样式管理，支持全局换肤
4. **丰富动画**：卡牌飞行、粒子特效、过渡动画增强游戏感
5. **着色器应用**：HSV调整、波纹效果等提升视觉品质
`─────────────────────────────────────────────────`
