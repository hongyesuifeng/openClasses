### 1.2 角色基类设计

```csharp
public abstract class CharacterModel : AbstractModel
{
    // === 基础属性 ===
    public abstract int StartingHp { get; }
    public abstract int StartingGold { get; }
    public virtual int MaxEnergy => 3;
    public virtual int BaseOrbSlotCount => 0;  // 法球槽位

    // === 卡组相关 ===
    public abstract IEnumerable<CardModel> StartingDeck { get; }
    public abstract CardPoolModel CardPool { get; }

    // === 遗物相关 ===
    public abstract RelicPoolModel RelicPool { get; }
    public abstract IReadOnlyList<RelicModel> StartingRelics { get; }

    // === 药水相关 ===
    public abstract PotionPoolModel PotionPool { get; }
    public virtual IReadOnlyList<PotionModel> StartingPotions => Array.Empty<PotionModel>();

    // === 解锁条件 ===
    protected virtual CharacterModel? UnlocksAfterRunAs => null;

    // === 视觉特性 ===
    public virtual CharacterAnimationDelays AnimationDelays => new();
    public virtual IEnumerable<AttackVfxType> AttackVfxTypes => new[] { AttackVfxType.Blunt };
}
```

## 3. 遗物系统

### 3.1 遗物基类设计

```csharp
public abstract class RelicModel : AbstractModel
{
    // === 基础属性 ===
    public abstract RelicRarity Rarity { get; }
    public Player? Owner { get; set; }
    public int StackCount { get; private set; }

    // === 特殊状态 ===
    public bool IsWax { get; set; }        // 蜡质遗物
    public bool IsMelted { get; set; }     // 已熔化
    public RelicStatus Status { get; set; } // 显示状态

    // === 触发方法（可重写）===
    // 战斗相关
    public virtual Task AfterCombatVictory(CombatRoom room) => Task.CompletedTask;
    public virtual Task BeforeTurnStart(...) => Task.CompletedTask;
    public virtual Task AfterAttack(AttackCommand cmd) => Task.CompletedTask;

    // 房间相关
    public virtual Task AfterRoomEntered(AbstractRoom room) => Task.CompletedTask;
    public virtual Task AfterActEntered() => Task.CompletedTask;

    // 获得时效果
    public virtual Task AfterObtained() => Task.CompletedTask;
    public virtual Task AfterRemoved() => Task.CompletedTask;
}
```

### 3.2 遗物稀有度

```csharp
public enum RelicRarity
{
    None,       // 无（如堆叠遗物Circlet）

    Uncommon,   // 不普通 - 较稀有
    Rare,       // 稀有 - 非常稀有

    Ancient     // 上古 - 特殊获取
```

### 3.3 遗物效果触发机制

遗物通过重写虚方法来响应游戏事件：

```csharp
// 示例：攻击类遗物
public sealed class RedSkull : RelicModel
{
    public override RelicRarity Rarity => RelicRarity.Common;

    // 生命值低于50%时增加力量
    public override decimal ModifyDamageAdditive(...)
    {
        if (Owner.Creature.CurrentHp <= Owner.Creature.MaxHp / 2)
            return 3;
        return 0;
    }
}

// 示例：抽牌类遗物
public sealed class Sundial : RelicModel
{
    private int _cardsPlayedThisTurn;

    public override Task AfterCardPlayed(CardModel card)
    {
        _cardsPlayedThisTurn++;
        if (_cardsPlayedThisTurn >= 3)
        {
            await CardCmd.DrawCards(2);
            _cardsPlayedThisTurn = 0;
        }
        return Task.CompletedTask;
    }
}
```

## 4. 特殊遗物类型

### 4.1 堆叠遗物

```csharp
// Circlet（头环）- 可无限堆叠的遗物
public sealed class Circlet : RelicModel
{
    public override RelicRarity Rarity => RelicRarity.None;

    // 每层增加2%伤害
    public override decimal ModifyDamageMultiplicative(...)
    {
        return 1m + (StackCount * 0.02m);
    }

    // 堆叠时增加层数
    public void Stack()
    {
        StackCount++;
    }
}
```

### 4.2 拾取即生效遗物

```csharp
public sealed class AlchemicalCoffer : RelicModel
{
    public override RelicRarity Rarity => RelicRarity.Ancient;

    // 获得时立即生效
    public override async Task AfterObtained()
    {
        // 增加4个药水槽
        Owner.MaxPotionSlots += 4;

        // 随机获得4个药水
        for (int i = 0; i < 4; i++)
        {
            var potion = await PotionCmd.GenerateRandomPotion();
            Owner.AddPotion(potion);
        }
    }
}
```

### 5.2 解锁状态管理

```csharp
public class UnlockState
{
    private readonly HashSet<ModelId> _unlockedCharacters;

    public bool IsCharacterAvailable(CharacterModel character)
    {
        // 检查是否已解锁
        if (_unlockedCharacters.Contains(character.Id))
            return true;

        // 检查解锁条件
        var requirement = character.UnlocksAfterRunAs;
        if (requirement == null)
            return true;  // 默认解锁

        return HasCompletedRunWith(requirement);
    }
}
```

## 7. 设计亮点总结

`★ Insight ─────────────────────────────────────`
**Slay the Spire 2 角色与遗物系统的设计亮点：**

1. **角色差异化**：每个角色有独特的初始牌组、遗物和特殊机制（法球、星星等）
2. **遗物触发系统**：通过虚方法重写实现灵活的效果绑定，无需硬编码事件监听
3. **动态变量复用**：遗物和卡牌共用 DynamicVar 系统，统一数值管理
4. **解锁进度**：通过角色解锁引导玩家尝试不同玩法
`─────────────────────────────────────────────────`
